Beginner 05

Largest of 3 Numbers

Logic: Compare each number using relational and logical operators.

Key concept: && means logical AND.

#include <stdio.h>

int main(void) {
    int a, b, c;

    printf("Enter three numbers: ");
    scanf("%d %d %d", &a, &b, &c);

    if (a >= b && a >= c)
        printf("Largest = %d", a);
    else if (b >= a && b >= c)
        printf("Largest = %d", b);

Beginner 06

Leap Year

Logic: A year is leap if divisible by 400, or divisible by 4 but not by 100.

Practice: Test years such as 2024, 1900 and 2000.

#include <stdio.h>

int main(void) {
    int year;

    printf("Enter year: ");
    scanf("%d", &year);

    if ((year % 400 == 0) ||
        (year % 4 == 0 && year % 100 != 0))
        printf("Leap Year");
    else
        printf("Not a Leap Year");

    return 0;
}
Intermediate 07

Calculator Using Switch

Logic: Read an operator and use switch to select an operation.

Viva: Why is break used? It stops execution from falling into the next case.

#include <stdio.h>

int main(void) {
    double a, b;
    char op;

    printf("Enter: number operator number: ");
    scanf("%lf %c %lf", &a, &op, &b);

    switch (op) {
        case '+':
            printf("Result = %.2f", a + b);
            break;
        case '-':
            printf("Result = %.2f", a - b);
            break;
        case '*':
            printf("Result = %.2f", a * b);
            break;
        case '/':
            if (b == 0)
                printf("Cannot divide by zero");
            else
                printf("Result = %.2f", a / b);
            break;
        default:
            printf("Invalid operator");
    }

    return 0;
}

Intermediate 08

Factorial

What is this? Factorial of n is the product of all positive integers from 1 to n.

Logic: Use a loop from 1 to n and multiply into an accumulator.

Example: 5! = 120

#include <stdio.h>

int main(void) {
    int n;
    unsigned long long fact = 1;

    printf("Enter a positive integer: ");
    scanf("%d", &n);

    if (n < 0) {
        printf("Factorial is not defined for negative integers.");
        return 0;
    }

    for (int i = 1; i <= n; i++)
        fact *= i;

    printf("%d! = %llu", n, fact);
    return 0;
}
Intermediate 09

Fibonacci Series

Logic: Each new term is the sum of the previous two terms.

Example: 0 1 1 2 3 5 8...

#include <stdio.h>

int main(void) {
    int n;
    long long a = 0, b = 1, next;

    printf("How many terms? ");
    scanf("%d", &n);

    for (int i = 0; i < n; i++) {
        printf("%lld ", a);
        next = a + b;
        a = b;
        b = next;
    }

    return 0;
}
Intermediate 10

Prime Number

Logic: A prime number greater than 1 has no divisor other than 1 and itself.

Tip: Test divisors only up to the square root of the number.

#include <stdio.h>

int main(void) {
    int n, isPrime = 1;

    printf("Enter a number: ");
    scanf("%d", &n);

    if (n < 2)
        isPrime = 0;

    for (int i = 2; i <= n / i && isPrime; i++) {
        if (n % i == 0)
            isPrime = 0;
    }

    if (isPrime)
        printf("Prime");
    else
        printf("Not Prime");

    return 0;
}
Intermediate 11

Palindrome Number

Logic: Reverse the number and compare it with the original.

Example: 121 is a palindrome because it reads the same backwards.

#include <stdio.h>

int main(void) {
    int n, original, reverse = 0, remainder;

    printf("Enter a number: ");
    scanf("%d", &n);

    original = n;

    while (n != 0) {
        remainder = n % 10;
        reverse = reverse * 10 + remainder;
        n /= 10;
    }

    if (original == reverse)
        printf("Palindrome");
    else
        printf("Not Palindrome");

    return 0;
}
Beginner 12

Sum of Digits

#include <stdio.h>

int main(void) {

    scanf("%d", &n);

    while (n > 0) {

        n /= 10;
    }

    printf("Sum of digits = %d", sum);
    return 0;
}

Intermediate 15

Array Sum and Average

Logic: Read array elements, add them, then divide the sum by the number of elements.

Concept: One-dimensional arrays and loops.

#include <stdio.h>

int main(void) {
    int a[100], n, sum = 0;

    printf("How many elements? ");
    scanf("%d", &n);

    if (n < 1 || n > 100) {
        printf("Invalid size");
        return 0;
    }

    for (int i = 0; i < n; i++)
        scanf("%d", &a[i]);

    for (int i = 0; i < n; i++)
        sum += a[i];

    printf("Sum = %d\n", sum);
    printf("Average = %.2f", (double)sum / n);

    return 0;
}
Intermediate 16

Maximum and Minimum in Array

Logic: Start with the first element as both maximum and minimum, then compare the remaining elements.

#include <stdio.h>

int main(void) {
    int a[100], n;

    printf("How many elements? ");
    scanf("%d", &n);

    if (n < 1 || n > 100) {
        printf("Invalid size");
        return 0;
    }

    for (int i = 0; i < n; i++)
        scanf("%d", &a[i]);

    int max = a[0];
    int min = a[0];

    for (int i = 1; i < n; i++) {
        if (a[i] > max) max = a[i];
        if (a[i] < min) min = a[i];
    }

    printf("Maximum = %d\nMinimum = %d", max, min);
    return 0;
}
Advanced 17

Matrix Addition

Logic: Add corresponding elements of two matrices using nested loops.

Concept: Two-dimensional arrays.

#include <stdio.h>

int main(void) {
    int a[2][2], b[2][2], c[2][2];

    printf("Enter matrix A (2 x 2):\n");
    for (int i = 0; i < 2; i++)
        for (int j = 0; j < 2; j++)
            scanf("%d", &a[i][j]);

    printf("Enter matrix B (2 x 2):\n");
    for (int i = 0; i < 2; i++)
        for (int j = 0; j < 2; j++)
            scanf("%d", &b[i][j]);

    printf("Result:\n");
    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++) {
            c[i][j] = a[i][j] + b[i][j];
            printf("%d ", c[i][j]);
        }
        printf("\n");
    }

    return 0;
}

Intermediate 18

String Length

Logic: Read a line and count characters until the null terminator.

Important: This version uses fgets() instead of unsafe gets().

#include <stdio.h>
#include <string.h>

int main(void) {
    char s[100];

    printf("Enter a string: ");
    fgets(s, sizeof(s), stdin);

    s[strcspn(s, "\n")] = '\0';

    printf("Length = %zu", strlen(s));
    return 0;
}
Intermediate 19

String Reverse

Logic: Find the last character and print the string backwards.

Important: This program does not depend on non-standard strrev().

#include <stdio.h>
#include <string.h>

int main(void) {
    char s[100];

    printf("Enter a string: ");
    fgets(s, sizeof(s), stdin);

    s[strcspn(s, "\n")] = '\0';

    printf("Reversed = ");
    for (int i = (int)strlen(s) - 1; i >= 0; i--)
        putchar(s[i]);

    return 0;
}

Functions & Pointers

Move from basic programs to reusable functions and memory concepts.

Advanced 20

Swap with Pointers

Logic: Pass addresses to a function so the function can change the original variables.

Viva: * dereferences a pointer and & obtains an address.

#include <stdio.h>

void swap(int *x, int *y) {
    int temp = *x;
    *x = *y;
    *y = temp;
}

int main(void) {
    int a = 5, b = 10;

    swap(&a, &b);

    printf("a = %d\nb = %d", a, b);
    return 0;
}
Intermediate 21

Swap Without Temporary Variable

Logic: Use arithmetic operations to exchange two integer values.

Note: A temporary variable is usually safer and easier to understand in real programs.

int main(void) {
    int a, b;

    printf("Enter two integers: ");
    scanf("%d %d", &a, &b);

    a = a + b;
    b = a - b;
    a = a - b;

    printf("After swap: %d %d", a, b);
    return 0;
}
Advanced 22

Factorial Using a Function

Logic: The function calls itself with a smaller value until it reaches the base case.

Concept: Recursion.

#include <stdio.h>

unsigned long long fact(int n) {
    if (n == 0)
        return 1;

    return (unsigned long long)n * fact(n - 1);
}

int main(void) {
    int n;

    printf("Enter a non-negative integer: ");
    scanf("%d", &n);

    if (n < 0)
        printf("Invalid input");
    else
        printf("%d! = %llu", n, fact(n));

    return 0;
}

Advanced 23

Student Structure

Logic: A structure groups related data such as roll number, name and marks.

Practical use: Structures are useful in student management systems.

#include <stdio.h>

struct Student {
    int roll;
    char name[50];
    float marks;
};

int main(void) {
    struct Student s;

    printf("Enter roll, name and marks: ");
    scanf("%d %49s %f", &s.roll, s.name, &s.marks);

    printf("\nRoll: %d", s.roll);
    printf("\nName: %s", s.name);
    printf("\nMarks: %.2f", s.marks);

    return 0;
}
Advanced 24

File Write

Logic: Open a file in write mode, write data, then close the file.

Important: Always check whether fopen() succeeded.

#include <stdio.h>

int main(void) {
    FILE *f = fopen("test.txt", "w");

    if (f == NULL) {
        printf("Could not open file.");
        return 1;
    }

    fprintf(f, "Hello Diploma");
    fclose(f);

    printf("File written successfully.");
    return 0;
}
Advanced 25

File Read

Logic: Open the file in read mode and display characters until end-of-file.

Important: fgetc() returns an int so it can represent EOF.

#include <stdio.h>

int main(void) {
    FILE *f = fopen("test.txt", "r");
    int ch;

    if (f == NULL) {
        printf("Could not open test.txt.");
        return 1;
    }

    while ((ch = fgetc(f)) != EOF)
        putchar(ch);

    fclose(f);
    return 0;
}
